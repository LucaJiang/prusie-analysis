"""Deterministic first-order carryover-balanced even-order Williams squares."""
import hashlib, random
SEED=20261023
def square(k):
    if k not in (2,4,8): raise ValueError('Williams order must be 2, 4 or 8')
    first=[0]
    for step in range(1,k):first.append((step+1)//2 if step%2 else k-step//2)
    return [[(value+r)%k for value in first] for r in range(k)]
def orders(case_id,backends,retained=8):
    k=len(backends)
    if retained not in (8,16) or retained%k:raise ValueError('Require 8 or 16 retained rounds')
    base=square(k)
    rng=random.Random(int.from_bytes(hashlib.sha256(f'{SEED}:Williams:{case_id}'.encode()).digest(),'big'))
    mapping=list(backends);rng.shuffle(mapping)
    rows=[]
    for _ in range(retained//k):
        block=list(range(k));rng.shuffle(block)
        rows.extend(base[r] for r in block)
    warmup=base[rng.randrange(k)]
    return [dict(call_index=i,backends=[mapping[s] for s in row]) for i,row in enumerate([warmup,*rows])]
